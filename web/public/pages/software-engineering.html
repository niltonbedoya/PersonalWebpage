@router.post("/chat", response_model=AgentVerdict)
async def dispatch_request(query: UserPrompt):
    # Perimeter guardrail < 150ms
    guard = await Centinela.evaluate(query.text)
    if not guard.allowed:
        return guard.redirect()

    # Dynamic zero-cost LLM routing
    agent = Nexus.route(query.intent)
    return await agent.execute(
        primary=Provider.GOOGLE_AI,
        fallback=Provider.GROQ
    )
